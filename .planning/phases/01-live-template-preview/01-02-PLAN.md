   - If function doesn't exist, import from existing project API or create stub
   - Pass AbortController.signal to fetch; catch AbortError and ignore (cancelled request)

3. **State Management**
   - useState for previewHtml (string, initially '')
   - useState for error (string | null, initially null)
   - useState for loading (boolean, initially false)
   - setLoading(true) when timer starts
   - setLoading(false) when generation completes

4. **Error Handling**
   - If buildPreviewHtml() throws, catch error
   - Set error state to error.message
   - Generate fallback error HTML: buildErrorHtml('Failed to generate preview: ' + error.message)
   - Log error to console (but don't throw)

5. **Cleanup**
   - useEffect cleanup function must:
     - Clear timer via clearTimeout()
     - Abort pending request via abortRef.current.abort()
   - Prevents memory leaks on unmount or rapid re-triggers

6. **Edge Cases**
   - If config is undefined or empty, set previewHtml to '' (no generation)
   - If templateId is undefined or empty, set previewHtml to '' (no generation)
   - If delay < 50ms, warn but allow (prevents accidental 0ms loops)

**Code structure:**
```javascript
import { useEffect, useRef, useState } from 'react';
import { buildPreviewHtml } from '../utils/template-preview-runtime.js';
import { buildErrorHtml } from '../utils/error-html.js'; // (import or create)

export function usePreviewDebounce(config, templateId, delay = 400) {
  const [previewHtml, setPreviewHtml] = useState('');
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(false);
  const timerRef = useRef(null);
  const abortRef = useRef(null);

  useEffect(() => {
    // Cleanup previous
    if (timerRef.current) clearTimeout(timerRef.current);
    if (abortRef.current) abortRef.current.abort();

    // Validation
    if (!config || !templateId) {
      setPreviewHtml('');
      setError(null);
      return;
    }

    setLoading(true);

    // New timer
    const abort = new AbortController();
    abortRef.current = abort;

    timerRef.current = setTimeout(async () => {
      try {
        // Fetch template files
        const files = await getTemplateFiles(templateId, { signal: abort.signal });

        // Generate preview
        const html = buildPreviewHtml(files, config);
        setPreviewHtml(html);
        setError(null);
      } catch (e) {
        if (e.name !== 'AbortError') {
          console.error('Preview generation failed:', e);
          setError(e.message);
          setPreviewHtml(buildErrorHtml(e.message));
        }
      } finally {
        setLoading(false);
      }
    }, delay);

    return () => {
      if (timerRef.current) clearTimeout(timerRef.current);
      if (abortRef.current) abortRef.current.abort();
    };
  }, [config, templateId, delay]);

  return { previewHtml, error, loading };
}
```

```javascript
import React, { useState, useRef } from 'react';
import { usePreviewDebounce } from '../../hooks/usePreviewDebounce.js';
import { AntiFingerprint } from '../../services/AntiFingerprint.js';

export function PreviewModal({ isOpen, onClose, config, templateId }) {
  const [viewport, setViewport] = useState('desktop');
  const [showFingerprint, setShowFingerprint] = useState(false);
  const [fingerprintedHtml, setFingerprintedHtml] = useState('');
  const [generatingFingerprint, setGeneratingFingerprint] = useState(false);
  const iframeRef = useRef(null);

  const { previewHtml, error: previewError, loading: previewLoading } =
    usePreviewDebounce(config, templateId);

  if (!isOpen) return null;

  const handleFingerprintToggle = async () => {
    if (!showFingerprint) {
      // Generate fingerprinted version
      if (!fingerprintedHtml) {
        setGeneratingFingerprint(true);
        try {
          const result = await AntiFingerprint.transform(
            previewHtml,
            '', // cssContent
            config.siteId
          );
          setFingerprintedHtml(result.html);
        } catch (e) {
          console.error('Fingerprint generation failed:', e);
          // Don't toggle; show error
          setGeneratingFingerprint(false);
          return;
        }
        setGeneratingFingerprint(false);
      }
    }
    setShowFingerprint(!showFingerprint);
  };

  const iframeWidth = viewport === 'mobile' ? '320px' : '1024px';
  const iframeHeight = viewport === 'mobile' ? '640px' : '768px';
  const displayHtml = showFingerprint ? fingerprintedHtml : previewHtml;

  return (
    <div className="modal-overlay" onClick={onClose}>
      <div className="modal-content" onClick={e => e.stopPropagation()}>
        <div className="modal-header">
          <h2>Live Preview</h2>
          <button onClick={onClose}>✕</button>
        </div>

        <div className="preview-controls">
          <button onClick={() => setViewport(viewport === 'mobile' ? 'desktop' : 'mobile')}>
            {viewport === 'mobile' ? '📱 Mobile' : '🖥️ Desktop'}
          </button>
          <button
            onClick={handleFingerprintToggle}
            disabled={generatingFingerprint}
          >
            {showFingerprint ? 'Show Original' : 'Show Fingerprinted'}
            {generatingFingerprint && ' (loading...)'}
          </button>
        </div>

        {(previewLoading || generatingFingerprint) && (
          <div className="preview-loading">Generating preview...</div>
        )}

        {(previewError || (showFingerprint && fingerprintedHtml && !displayHtml)) && (
          <div className="preview-error">{previewError || 'Preview generation failed'}</div>
        )}

        <iframe
          ref={iframeRef}
          sandbox="allow-scripts allow-same-origin allow-forms"
          srcDoc={displayHtml}
          style={{
            width: iframeWidth,
            height: iframeHeight,
            border: '1px solid #ddd',
            borderRadius: '8px',
            overflow: 'auto',
          }}
          title="Preview"
        />
      </div>
    </div>
  );
}
```
