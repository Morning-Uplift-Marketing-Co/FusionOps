   - Error handling already in usePreviewDebounce; just verify message is helpful

   - If previewHtml is extremely large, buildPreviewHtml() may be slow
   - Debounce already handles this (400ms delay)
   - DiffViewer already truncates large diffs (maxLines=100)
   - Consider: Show performance warning if preview takes >1s: "Large template; preview may take a moment"

   - If AntiFingerprint.transform() throws error
   - Catch error in handleFingerprintToggle
   - Set fingerprintError state
   - Show error message: "Could not generate fingerprint preview: {error.message}"
   - Recommend: "This may be a temporary issue; try again"

5. **siteId Required but Missing**
   - If Wizard Step 5 is reached without siteId being assigned
   - This is a workflow issue (siteId should be assigned before review)
   - Document: "siteId must be generated before Step 5"
   - Graceful fallback: Generate temporary siteId (see case 1)

```javascript
// Helper: Generate temporary siteId if missing
function generateTemporarySiteId(templateId) {
  const crypto = require('crypto');
  const hash = crypto.createHash('sha256')
    .update(`${templateId}-${Date.now()}`)
    .digest('hex')
    .substring(0, 16);
  return `temp-${hash}`;
}

// In PreviewModal handleFingerprintToggle:
const handleFingerprintToggle = async () => {
  if (!showFingerprint) {
    if (!fingerprintedHtml) {
      setGeneratingFingerprint(true);
      setFingerprintError(null);

      try {
        // Check siteId
        const siteId = config.siteId || generateTemporarySiteId(templateId);
        if (!config.siteId) {
          console.warn('siteId not available; using temporary ID for preview');
        }

        const result = await AntiFingerprint.transform(
          previewHtml,
          '',
          siteId
        );
        setFingerprintedHtml(result.html);
        setActivePreviewTab('compare');
      } catch (e) {
        console.error('Fingerprint generation failed:', e);
        setFingerprintError(`Could not generate fingerprint preview: ${e.message}`);
      } finally {
        setGeneratingFingerprint(false);
      }
    } else {
      setActivePreviewTab('compare');
    }
  }
};
```
