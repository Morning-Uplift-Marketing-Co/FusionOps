```javascript
export function PreviewModal({ isOpen, onClose, config, templateId }) {
  const [activePreviewTab, setActivePreviewTab] = useState('preview');
  const [viewport, setViewport] = useState('desktop');
  const [fingerprintedHtml, setFingerprintedHtml] = useState('');
  const [generatingFingerprint, setGeneratingFingerprint] = useState(false);
  const [fingerprintError, setFingerprintError] = useState(null);
  const iframeRef = useRef(null);

  const { previewHtml, error: previewError, loading: previewLoading } =
    usePreviewDebounce(config, templateId);

  if (!isOpen) return null;

  const handleSwitchToCompare = async () => {
    if (activePreviewTab === 'compare') {
      setActivePreviewTab('preview');
      return;
    }

    // Switch to compare tab
    if (!fingerprintedHtml && !generatingFingerprint) {
      setGeneratingFingerprint(true);
      setFingerprintError(null);
      try {
        const result = await AntiFingerprint.transform(
          previewHtml,
          '', // cssContent
          config.siteId
        );
        setFingerprintedHtml(result.html);
        setActivePreviewTab('compare');
      } catch (e) {
        console.error('Fingerprint generation failed:', e);
        setFingerprintError(e.message);
      } finally {
        setGeneratingFingerprint(false);
      }
    } else if (fingerprintedHtml) {
      setActivePreviewTab('compare');
    }
  };

  const iframeWidth = viewport === 'mobile' ? '320px' : '1024px';
  const iframeHeight = viewport === 'mobile' ? '640px' : '768px';

  return (
    <div className="modal-overlay" onClick={onClose}>
      <div className="modal-content" onClick={e => e.stopPropagation()}>
        <div className="modal-header">
          <h2>Live Preview</h2>
          <button onClick={onClose}>✕</button>
        </div>

        {/* Tab buttons */}
        <div className="modal-tabs">
          <button
            className={`tab-button ${activePreviewTab === 'preview' ? 'active' : ''}`}
            onClick={() => setActivePreviewTab('preview')}
          >
            Live Preview
          </button>
          <button
            className={`tab-button ${activePreviewTab === 'compare' ? 'active' : ''}`}
            onClick={handleSwitchToCompare}
            disabled={generatingFingerprint}
          >
            Fingerprint Comparison
            {generatingFingerprint && ' (loading...)'}
          </button>
        </div>

        {/* Preview Tab */}
        {activePreviewTab === 'preview' && (
          <>
            <div className="preview-controls">
              <button onClick={() => setViewport(viewport === 'mobile' ? 'desktop' : 'mobile')}>
                {viewport === 'mobile' ? '📱 Mobile' : '🖥️ Desktop'}
              </button>
            </div>

            {previewLoading && <div className="preview-loading">Generating preview...</div>}
            {previewError && <div className="preview-error">{previewError}</div>}

            <iframe
              ref={iframeRef}
              sandbox="allow-scripts allow-same-origin allow-forms"
              srcDoc={previewHtml}
              style={{
                width: iframeWidth,
                height: iframeHeight,
                border: '1px solid #ddd',
                borderRadius: '8px',
                overflow: 'auto',
              }}
              title="Preview"
            />
          </>
        )}

        {/* Fingerprint Comparison Tab */}
        {activePreviewTab === 'compare' && (
          <>
            {fingerprintError && (
              <div className="preview-error">{fingerprintError}</div>
            )}

            {fingerprintedHtml ? (
              <DiffViewer
                preHtml={previewHtml}
                postHtml={fingerprintedHtml}
                layout="side-by-side"
                maxLines={100}
                showSummary={true}
              />
            ) : (
              <div className="preview-loading">No comparison available</div>
            )}
          </>
        )}
      </div>
    </div>
  );
}
```
