**Implementation requirements:**

1. **Import diff-match-patch**
   - `import diff_match_patch from 'diff-match-patch'`
   - Or use CommonJS: `const diff_match_patch = require('diff-match-patch');` if ESM issues
   - Verify npm package installed: `npm install diff-match-patch`

2. **Generate Diffs**
   - Create new diff_match_patch() instance
   - Call .diff_main(preHtml, postHtml)
   - Returns array of [operation_type, text] tuples:
     - 0 = unchanged
     - 1 = added (in postHtml)
     - -1 = removed (in preHtml)
   - Call .diff_cleanupSemantic(diffs) to merge adjacent changes

3. **Calculate Summary**
   - Count total added characters: sum of text lengths where operation=1
   - Count total removed characters: sum of text lengths where operation=-1
   - Count unchanged characters: sum of text lengths where operation=0
   - Count change operations: (diffs where operation !== 0).length
   - Return summary object: {added, removed, unchanged, changeCount}

4. **Options Support**
   - options.ignoreWhitespace: boolean (default false)
   - If true, strip leading/trailing whitespace from pre/post before diff
   - options.maxDiffLines: number (default Infinity)
   - If diffs > maxDiffLines, truncate to first maxDiffLines/2 + last maxDiffLines/2
   - options.cleanupSemantic: boolean (default true)
   - If true, call diff_cleanupSemantic() for readability

5. **Return Format**
   ```javascript
   {
     diffs: [[0, '<div'], [1, ' class="fp-abc"'], [0, '>Test</div>'], ...],
     summary: {
       added: 18,      // characters added
       removed: 0,     // characters removed
       unchanged: 120, // characters unchanged
       changeCount: 1  // number of change operations
     }
   }
   ```

6. **Edge Cases**
   - If preHtml === postHtml, return {diffs: [], summary: {added: 0, removed: 0, unchanged: preHtml.length, changeCount: 0}}
   - If preHtml or postHtml is empty string, handle gracefully (treat as 0-byte content)
   - If preHtml/postHtml is null or undefined, throw error with clear message: "preHtml and postHtml must be strings"

**Code structure:**
```javascript
import diff_match_patch from 'diff-match-patch';

export function generateHtmlDiff(preHtml, postHtml, options = {}) {
  if (typeof preHtml !== 'string' || typeof postHtml !== 'string') {
    throw new Error('preHtml and postHtml must be strings');
  }

  const {
    ignoreWhitespace = false,
    maxDiffLines = Infinity,
    cleanupSemantic = true,
  } = options;

  let pre = preHtml;
  let post = postHtml;

  // Optionally ignore whitespace
  if (ignoreWhitespace) {
    pre = pre.trim();
    post = post.trim();
  }

  // Check if identical
  if (pre === post) {
    return {
      diffs: [],
      summary: {
        added: 0,
        removed: 0,
        unchanged: pre.length,
        changeCount: 0,
      },
    };
  }

  // Generate diffs
  const dmp = new diff_match_patch();
  const diffs = dmp.diff_main(pre, post);

  if (cleanupSemantic) {
    dmp.diff_cleanupSemantic(diffs);
  }

  // Truncate if needed
  let finalDiffs = diffs;
  if (diffs.length > maxDiffLines) {
    const half = Math.floor(maxDiffLines / 2);
    finalDiffs = [
      ...diffs.slice(0, half),
      [0, `... (${diffs.length - maxDiffLines} diffs omitted) ...`],
      ...diffs.slice(-half),
    ];
  }

  // Calculate summary
  const summary = {
    added: 0,
    removed: 0,
    unchanged: 0,
    changeCount: 0,
  };

  diffs.forEach(([op, text]) => {
    if (op === 1) summary.added += text.length;
    else if (op === -1) summary.removed += text.length;
    else summary.unchanged += text.length;

    if (op !== 0) summary.changeCount++;
  });

  return { diffs: finalDiffs, summary };
}
```

**Code structure:**
```javascript
import React, { useState, useMemo } from 'react';
import { generateHtmlDiff } from '../../utils/html-diff.js';

export function DiffViewer({
  preHtml,
  postHtml,
  diffs: propDiffs,
  layout = 'side-by-side',
  maxLines = 50,
  showSummary = true,
}) {
  const [activeTab, setActiveTab] = useState('pre');

  // Compute diffs if not provided
  const { diffs, summary } = useMemo(() => {
    if (propDiffs) {
      // Use provided diffs and recompute summary
      const added = propDiffs.reduce((sum, [op, text]) => sum + (op === 1 ? text.length : 0), 0);
      const removed = propDiffs.reduce((sum, [op, text]) => sum + (op === -1 ? text.length : 0), 0);
      const unchanged = propDiffs.reduce((sum, [op, text]) => sum + (op === 0 ? text.length : 0), 0);
      return {
        diffs: propDiffs,
        summary: { added, removed, unchanged, changeCount: propDiffs.filter(d => d[0] !== 0).length },
      };
    }

    return generateHtmlDiff(preHtml, postHtml, { maxDiffLines: maxLines });
  }, [preHtml, postHtml, propDiffs, maxLines]);

  if (!diffs || diffs.length === 0) {
    return <div className="diff-empty">No changes detected</div>;
  }

  const renderDiffSegment = () => (
    <code>
      {diffs.map((diff, idx) => {
        const [op, text] = diff;
        if (op === 0) return <span key={idx}>{text}</span>;
        if (op === 1) return <span key={idx} className="diff-added">{text}</span>;
        if (op === -1) return <span key={idx} className="diff-removed">{text}</span>;
      })}
    </code>
  );

  return (
    <div className="diff-viewer">
      {showSummary && (
        <div className="diff-summary">
          <p>
            {summary.added > 0 && <span className="summary-added">+{summary.added}</span>}
            {summary.removed > 0 && <span className="summary-removed">-{summary.removed}</span>}
            <span className="summary-unchanged">{summary.unchanged} unchanged</span>
            <span className="summary-changes">{summary.changeCount} changes</span>
          </p>
        </div>
      )}

      {layout === 'tabbed' ? (
        <div className="diff-tabs">
          <div className="diff-tab-buttons">
            <button
              className={activeTab === 'pre' ? 'active' : ''}
              onClick={() => setActiveTab('pre')}
            >
              Original
            </button>
            <button
              className={activeTab === 'post' ? 'active' : ''}
              onClick={() => setActiveTab('post')}
            >
              Fingerprinted
            </button>
          </div>
          <div className="diff-tab-content">
            {activeTab === 'pre' ? (
              <pre className="diff-pre">{preHtml}</pre>
            ) : (
              <pre className="diff-post">{postHtml}</pre>
            )}
          </div>
        </div>
      ) : (
        // Side-by-side layout
        <div className="diff-columns">
          <div className="diff-column">
            <h4>Original</h4>
            <pre className="diff-pre">{preHtml}</pre>
          </div>
          <div className="diff-column">
            <h4>Fingerprinted</h4>
            <pre className="diff-post">{renderDiffSegment()}</pre>
          </div>
        </div>
      )}
    </div>
  );
}
```
