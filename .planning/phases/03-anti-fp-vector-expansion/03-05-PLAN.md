#!/usr/bin/env node

import fs from 'fs';
import path from 'path';
import { TemplateBuilder } from '../src/adapters/TemplateBuilder.js';

/**
 * Phase 3 Alpha Test 2 Deployment
 * Deploys 5-10 test domains with varying vector combinations
 */

async function deployDomains() {
  const domainsFile = '.planning/alpha-test/domains.json';
  const manifest = JSON.parse(fs.readFileSync(domainsFile, 'utf8'));

  const results = [];

  for (const domainConfig of manifest.domains) {
    console.log(`Deploying ${domainConfig.siteId}...`);

    try {
      // Build HTML with selected vectors
      const templatePath = `templates/${domainConfig.template}/index.html`;
      const templateContent = fs.readFileSync(templatePath, 'utf8');

      const result = await TemplateBuilder.orchestrate(
        templateContent,
        domainConfig.siteId,
        {
          vectors: domainConfig.vectors
        }
      );

      // Record deployment
      results.push({
        siteId: domainConfig.siteId,
        domain: domainConfig.domain,
        deployedAt: new Date().toISOString(),
        vectors: domainConfig.vectors,
        buildSuccess: true
      });

      // Deploy to staging (stubbed - assumes vercel/netlify CLI)
      console.log(`✓ ${domainConfig.siteId} built successfully`);
    } catch (err) {
      console.error(`✗ ${domainConfig.siteId} failed:`, err.message);
      results.push({
        siteId: domainConfig.siteId,
        domain: domainConfig.domain,
        deployedAt: new Date().toISOString(),
        vectors: domainConfig.vectors,
        buildSuccess: false,
        error: err.message
      });
    }
  }

  // Output deployment record
  const record = {
    timestamp: new Date().toISOString(),
    domainsDeployed: results.filter(r => r.buildSuccess).length,
    totalDomains: results.length,
    results
  };

  console.log('\n=== Deployment Summary ===');
  console.log(`Deployed: ${record.domainsDeployed}/${record.totalDomains}`);
  console.log(JSON.stringify(record, null, 2));

  return record;
}

deployDomains().catch(console.error);

#!/usr/bin/env node

import { TemplateBuilder } from '../src/adapters/TemplateBuilder.js';
import fs from 'fs';

/**
 * Performance benchmark for Phase 3 vector pipeline
 * Test at 20, 40, 50+ concurrent builds
 */

async function benchmarkConcurrentBuilds(concurrencyLevel) {
  const sampleHtml = fs.readFileSync('templates/vite-react/index.html', 'utf8');
  const startTime = Date.now();
  const startMemory = process.memoryUsage().heapUsed / 1024 / 1024; // MB

  const promises = [];
  for (let i = 0; i < concurrencyLevel; i++) {
    promises.push(
      TemplateBuilder.orchestrate(sampleHtml, `site-${i}`, {
        vectors: {
          obfuscate: true,
          networkJitter: true,
          eventRandomization: true
        }
      })
    );
  }

  await Promise.all(promises);

  const endTime = Date.now();
  const endMemory = process.memoryUsage().heapUsed / 1024 / 1024; // MB

  return {
    concurrency: concurrencyLevel,
    totalTime: endTime - startTime,
    avgTimePerBuild: (endTime - startTime) / concurrencyLevel,
    memoryUsed: endMemory - startMemory,
    memoryPeakPercent: (endMemory / 1024) * 100 // Assume 1GB total
  };
}

async function runBenchmarks() {
  console.log('=== Phase 3 Performance Benchmarking ===\n');

  const levels = [20, 40, 50];
  const results = [];

  for (const level of levels) {
    console.log(`Testing ${level} concurrent builds...`);
    const result = await benchmarkConcurrentBuilds(level);
    results.push(result);

    console.log(`  Total time: ${result.totalTime}ms`);
    console.log(`  Avg per build: ${result.avgTimePerBuild.toFixed(0)}ms`);
    console.log(`  Memory delta: ${result.memoryUsed.toFixed(2)}MB`);
    console.log(`  Peak memory: ${result.memoryPeakPercent.toFixed(1)}%\n`);

    // Alert if peak memory >90%
    if (result.memoryPeakPercent > 90) {
      console.warn(`  ⚠️  Peak memory >90%! Consider implementing build queue.`);
    }
  }

  // Summary
  console.log('=== Benchmark Summary ===');
  console.log(`Target overhead: <15% increase from baseline`);
  console.log(`Target memory: <90% peak usage\n`);

  const safe = results.every(r => r.memoryPeakPercent < 90);
  console.log(safe ? '✓ All tests passed' : '✗ Memory warning - queue needed');

  // Output results
  fs.writeFileSync(
    '.planning/alpha-test/benchmark-results.json',
    JSON.stringify({ timestamp: new Date().toISOString(), results }, null, 2)
  );
}

runBenchmarks().catch(console.error);

function analyzeMonitoringData() {
  const lines = fs.readFileSync('.planning/alpha-test/daily-monitoring.jsonl', 'utf8')
    .trim()
    .split('\n')
    .map(l => JSON.parse(l));

  const analysis = {
    totalDays: lines.length,
    detectionTimeline: {},
    stillActive: []
  };

  // Group by domain, track first detection
  for (const day of lines) {
    for (const domain of day.domains) {
      if (!analysis.detectionTimeline[domain.siteId]) {
        analysis.detectionTimeline[domain.siteId] = {
          firstDetected: domain.detected ? day.date : null,
          daysToDetection: null
        };
      }
      if (domain.detected && !analysis.detectionTimeline[domain.siteId].daysToDetection) {
        analysis.detectionTimeline[domain.siteId].daysToDetection =
          (new Date(day.date) - new Date(lines[0].date)) / (1000 * 60 * 60 * 24);
      }
    }
  }

  // Count still-active at day 14
  const lastDay = lines[lines.length - 1];
  for (const domain of lastDay.domains) {
    if (!domain.detected) {
      analysis.stillActive.push(domain.siteId);
    }
  }

  // Calculate success percentages
  analysis.detectionRate = (lastDay.domains.filter(d => d.detected).length / lastDay.domains.length) * 100;
  analysis.stillActivePercent = (analysis.stillActive.length / lastDay.domains.length) * 100;

  return analysis;
}
