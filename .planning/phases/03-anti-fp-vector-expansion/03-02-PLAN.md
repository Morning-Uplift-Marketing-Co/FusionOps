```javascript
import crypto from 'crypto';
import seedrandom from 'seedrandom';
import terser from 'terser';

export class JavaScriptObfuscator {
  /**
   * Transform HTML by obfuscating inline scripts
   * @param {string} htmlContent - Raw HTML with inline scripts
   * @param {string} siteId - Deployment site ID (for seeding)
   * @param {Object} options - Configuration {level: 'moderate'|'aggressive'}
   * @returns {Promise<{html: string, scripts: Array, sourceMaps: Map, obfuscated: boolean}>}
   */
  static async transform(htmlContent, siteId, options = {}) {
    const { level = 'moderate' } = options;

    // Deterministic seeding: same siteId always produces same output
    const seed = crypto.createHash('sha256')
      .update(siteId + 'js-obfuscation')
      .digest('hex');
    // Note: seedrandom used for later enhancements; terser determinism comes from config

    // Extract inline scripts from HTML
    const scriptRegex = /<script(?!\s+src)(?!\s+type=[\"']module[\"'])[^>]*>([\s\S]*?)<\/script>/gi;
    let transformedHtml = htmlContent;
    const scripts = [];
    const sourceMaps = new Map();

    let match;
    let scriptIndex = 0;

    while ((match = scriptRegex.exec(htmlContent)) !== null) {
      const originalCode = match[1].trim();
      if (!originalCode) continue;

      try {
        // Terser configuration
        const terserConfig = {
          compress: {
            drop_console: false,  // Keep console for debugging
            drop_debugger: true,
            passes: level === 'aggressive' ? 3 : 1,
            pure_getters: true
          },
          mangle: {
            eval: true,
            properties: false,  // CRITICAL: preserve object properties for tracking data-*
            keep_fnames: false,
            toplevel: true,
            // Reserved names: common event handlers that shouldn't be mangled
            reserved: [
              'onclick', 'onsubmit', 'onchange', 'onload', 'onerror',
              'handleSubmit', 'handleClick', 'handleChange',
              'addEventListener', 'removeEventListener',
              'fetch', 'sendBeacon', 'XMLHttpRequest'
            ]
          },
          output: {
            comments: false,
            beautify: false
          },
          sourceMap: {
            content: 'inline',
            url: `inline-${scriptIndex}.js.map`
          }
        };

        // Minify with terser
        const result = await terser.minify(originalCode, terserConfig);

        if (result.error) {
          throw new Error(`Terser error: ${result.error.message}`);
        }

        // Store obfuscated code and source map
        const obfuscatedCode = result.code;
        scripts.push({
          index: scriptIndex,
          original: originalCode,
          obfuscated: obfuscatedCode,
          level: level
        });

        if (result.map) {
          sourceMaps.set(scriptIndex, result.map);
        }

        // Replace script content in HTML
        const scriptTag = htmlContent.substring(match.index, match.index + match[0].length);
        const newScriptTag = scriptTag.replace(originalCode, obfuscatedCode);
        transformedHtml = transformedHtml.replace(scriptTag, newScriptTag);

        scriptIndex++;
      } catch (err) {
        console.error(`Failed to obfuscate script ${scriptIndex}:`, err.message);
        throw err;
      }
    }

    return {
      html: transformedHtml,
      scripts: scripts,
      sourceMaps: sourceMaps,
      obfuscated: scripts.length > 0
    };
  }
}
```

   - Return both original + obfuscated for comparison/debugging
   - Error handling: throw on terser.error
