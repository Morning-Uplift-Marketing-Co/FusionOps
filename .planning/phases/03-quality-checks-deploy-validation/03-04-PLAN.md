import { TemplateBuilder } from '../../build/TemplateBuilder.js';
import { AntiFingerprint } from '../../fingerprint/AntiFingerprint.js';
import { QualityChecker } from '../QualityChecker.js';

describe('Quality Check E2E Pipeline', () => {
  test('Full pipeline: detect format → build → fingerprint → quality checks → deploy-ready', async () => {
    // Mock build output (simulating Phase 2 TemplateBuilder result)
    const mockBuildOutput = {
      success: true,
      outputDir: '/tmp/build-output',
      html: \`<html>
        <head>
          <meta name="viewport" content="width=device-width, initial-scale=1">
          <script src="https://www.googletagmanager.com/gtag/js?id=G-XXXXXXXXXX"></script>
          <script>gtag('config', 'G-XXXXXXXXXX');</script>
        </head>
        <body>
          <input name="gclid">
          Brand: INJECTED_VALUE
        </body>
      </html>\`,
      css: 'body { color: red; }',
    };

    // Step 1: Mock Phase 2 build
    jest.spyOn(TemplateBuilder, 'identifyFramework').mockReturnValue('astro');
    jest.spyOn(TemplateBuilder, 'selectBuilder').mockReturnValue({
      build: jest.fn().mockResolvedValue(mockBuildOutput),
    });

    // Step 2: Run anti-fingerprinting (Phase 2)
    const { html: fingerprintedHtml, css: fingerprintedCss } =
      await AntiFingerprint.transform(
        mockBuildOutput.html,
        mockBuildOutput.css,
        'test-site-id'
      );

    // Step 3: Run quality checks (Phase 3) ← NEW
    const qualityResults = await QualityChecker.validatePreDeploy(
      fingerprintedHtml,
      fingerprintedCss,
      {
        trackingConfig: { required: true },
        lighthouseConfig: { timeout: 5000 },
      }
    );

    // Verify all gates passed
    expect(qualityResults.passed).toBe(true);
    expect(qualityResults.checks.length).toBeGreaterThanOrEqual(4); // At least 4 checks
    expect(qualityResults.criticalFailures).toHaveLength(0);
    expect(qualityResults.summary.total).toBeGreaterThan(0);
  });

  test('Pipeline blocks deploy if viewport missing', async () => {
    const htmlWithoutViewport = \`<html>
      <head>
        <title>Missing Viewport</title>
      </head>
      <body>Test</body>
    </html>\`;

    const results = await QualityChecker.validatePreDeploy(
      htmlWithoutViewport,
      '',
      { lighthouseConfig: { timeout: 5000 } }
    );

    expect(results.passed).toBe(false);
    expect(results.criticalFailures.some((f) => f.id === 'QUAL-01')).toBe(true);
  });

  test('Pipeline blocks deploy if Astro leaks detected', async () => {
    const htmlWithLeaks = \`<html>
      <head>
        <meta name="viewport" content="width=device-width">
      </head>
      <body>
        <script>
          const brand = import.meta.env.PUBLIC_BRAND || "Default";
        </script>
      </body>
    </html>\`;

    const results = await QualityChecker.validatePreDeploy(
      htmlWithLeaks,
      '',
      { lighthouseConfig: { timeout: 5000 } }
    );

    expect(results.passed).toBe(false);
    expect(results.criticalFailures.some((f) => f.id === 'QUAL-03')).toBe(true);
  });

  test('Quality checks return detailed error messages with fix suggestions', async () => {
    const htmlWithoutTracking = \`<html>
      <head>
        <meta name="viewport" content="width=device-width">
      </head>
      <body>No tracking</body>
    </html>\`;

    const results = await QualityChecker.validatePreDeploy(
      htmlWithoutTracking,
      '',
      {
        trackingConfig: { required: true },
        lighthouseConfig: { timeout: 5000 },
      }
    );

    // Should fail due to missing tracking
    expect(results.passed).toBe(false);

    // Errors should have fix suggestions
    const failures = results.criticalFailures;
    failures.forEach((f) => {
      expect(f.message).toBeDefined();
      // Optional: expect(f.fix).toBeDefined(); // Some failures may not have fixes
    });
  });
});
